// ---------- menu & setup ----------
function onOpen() {
  SpreadsheetApp.getUi().createMenu('Бренд-менеджер')
    .addItem('Ключ подключения', 'showKey')
    .addItem('Первичная настройка', 'setup')
    .addToUi();
}

function setup() {
  var props = PropertiesService.getScriptProperties();
  if (!props.getProperty('KEY')) props.setProperty('KEY', Utilities.getUuid().replace(/-/g, ''));
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  props.setProperty('SS_ID', ss.getId());
  metaSheet_(ss);
  if (!ss.getSheetByName(SUMMARY)) {
    var sum = ss.insertSheet(SUMMARY, 0);
    sum.getRange('A1').setValue('Сводка появится после подключения сайта').setFontWeight('bold');
  }
  showKey();
}

function showKey() {
  var key = PropertiesService.getScriptProperties().getProperty('KEY');
  var msg = key ? 'Ключ подключения для сайта:\n\n' + key + '\n\nВставьте его на сайте: Настройки → Google Таблица.' : 'Сначала выполните «Первичная настройка».';
  Logger.log(msg);
  try { SpreadsheetApp.getUi().alert(msg); } catch (e) {}
}

// Any manual edit bumps the change stamp so the site knows to pull.
function onEdit(e) {
  try { if (e && e.range && e.range.getSheet().getName() !== META) bump_(e.source); } catch (err) {}
}

// ---------- web app ----------
function doPost(e) {
  var p = {};
  try { p = JSON.parse(e.postData.contents); } catch (x) {}
  return handle_(p);
}
function doGet(e) { return handle_(e.parameter || {}); }

function handle_(p) {
  var out;
  try {
    var key = PropertiesService.getScriptProperties().getProperty('KEY');
    if (!key || p.key !== key) throw new Error('Неверный ключ подключения');
    var ss = spreadsheet_();
    if (p.action === 'ping') out = { ok: true, url: ss.getUrl(), name: ss.getName() };
    else if (p.action === 'pull') out = pull_(ss, p.since);
    else if (p.action === 'push') {
      var lock = LockService.getScriptLock();
      lock.waitLock(25000);
      try { out = push_(ss, p); } finally { lock.releaseLock(); }
    } else throw new Error('Неизвестное действие');
  } catch (err) {
    out = { ok: false, error: String(err && err.message || err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function spreadsheet_() {
  var ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  if (!ss) ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('SS_ID'));
  return ss;
}

// ---------- meta ----------
function metaSheet_(ss) {
  var sh = ss.getSheetByName(META);
  if (!sh) {
    sh = ss.insertSheet(META);
    sh.getRange('A1:B1').setValues([['stamp', String(Date.now())]]);
    sh.getRange('A3:G3').setValues([['id', 'gid', 'name', 'capIng', 'capPack', 'capOnce', 'version']]);
    sh.hideSheet();
  }
  return sh;
}
function stamp_(ss) { return String(metaSheet_(ss).getRange('B1').getValue()); }
function bump_(ss) { metaSheet_(ss).getRange('B1').setValue(String(Date.now())); }
function readMeta_(ss) {
  var sh = metaSheet_(ss), last = sh.getLastRow();
  if (last < 4) return [];
  return sh.getRange(4, 1, last - 3, 7).getValues().filter(function (r) { return r[0]; }).map(function (r) {
    return { id: String(r[0]), gid: Number(r[1]), name: String(r[2]), capIng: Number(r[3]), capPack: Number(r[4]), capOnce: Number(r[5]), version: Number(r[6]) };
  });
}
function writeMeta_(ss, list) {
  var sh = metaSheet_(ss), last = sh.getLastRow();
  if (last >= 4) sh.getRange(4, 1, last - 3, 7).clearContent();
  if (list.length) sh.getRange(4, 1, list.length, 7).setValues(list.map(function (m) { return [m.id, m.gid, m.name, m.capIng, m.capPack, m.capOnce, m.version]; }));
}
function sheetByGid_(ss, gid) {
  var all = ss.getSheets();
  for (var i = 0; i < all.length; i++) if (all[i].getSheetId() === gid) return all[i];
  return null;
}

  ['acquiringPct', 'logisticsCoef', 'lastMilePct', 'processing', 'acceptance'].forEach(function (fld) { sh.getRange('B' + row(fld)).setBackground(C_CALC).setFontColor('#9A9A9A'); });
  sh.getRangeList(['F' + L.ing0 + ':F' + L.ingT, 'F' + L.pack0 + ':F' + L.packT, 'H' + L.ing0 + ':H' + L.ingT, 'H' + L.pack0 + ':H' + L.packT, 'F' + L.once0 + ':F' + L.onceT,
    'B' + L.prodU, 'B' + c + ':B' + (c + 10), 'B' + L.mkUnit, 'B' + L.res0 + ':E' + (L.res0 + 10)]).setNumberFormat(F_RUB2);
  sh.getRangeList(['G' + L.ing0 + ':G' + L.ingT, 'G' + L.pack0 + ':G' + L.packT, 'B' + L.onceT, 'B' + (c + 11) + ':B' + (c + 13), 'B' + (L.res0 + 12) + ':E' + (L.res0 + 12)]).setNumberFormat(F_RUB0);
  sh.getRangeList(['B' + (L.res0 + 11) + ':E' + (L.res0 + 11), 'B' + (L.res0 + 13) + ':E' + (L.res0 + 13), 'B' + L.ingT]).setNumberFormat(F_PCT);
  sh.getRangeList(['B' + (c + 14), 'B' + (L.res0 + 14) + ':E' + (L.res0 + 14)]).setNumberFormat(F_INT);
  sh.getRangeList(['A' + (c + 10) + ':B' + (c + 10), 'A' + (L.res0 + 10) + ':E' + (L.res0 + 10)]).setFontWeight('bold').setBackground(C_KEY);

  // validation
  var list = function (vals) { return SpreadsheetApp.newDataValidation().requireValueInList(vals, true).setAllowInvalid(false).build(); };
  sh.getRange('C6').setDataValidation(list(['мл', 'г']));
  sh.getRange('B14').insertCheckboxes();
  sh.getRange('D' + L.ing0 + ':D' + (L.ingT - 1)).setDataValidation(list(CURRENCIES));
  sh.getRange('C' + L.pack0 + ':C' + (L.packT - 1)).setDataValidation(list(CURRENCIES));
  sh.getRange('D' + L.once0 + ':D' + (L.onceT - 1)).setDataValidation(list(STAGES.map(function (s) { return s[1]; })));
  sh.getRange('B' + L.taxMode).setDataValidation(list(TAX_MODES.map(function (s) { return s[1]; })));
  sh.getRange('B' + L.salesMode).setDataValidation(list(SALES_MODES.map(function (s) { return s[1]; })));
  sh.getRange('B' + L.in0 + ':E' + L.in0).insertCheckboxes();
  sh.setFrozenRows(0);
}

// ---------- push: site → sheet ----------
function capsFor_(inp) {
  return { ing: Math.max(15, (inp.ingredients || []).length + 3), pack: Math.max(10, (inp.packaging || []).length + 3), once: Math.max(10, (inp.oneOff || []).length + 3) };
}
function clean_(v) { return typeof v === 'string' ? v.replace(/^[=+\-@]+/, '') : v; }
function label_(pairs, id) { for (var i = 0; i < pairs.length; i++) if (pairs[i][0] === id) return pairs[i][1]; return pairs[0][1]; }
function idOf_(pairs, label) { label = String(label || '').trim().toLowerCase(); for (var i = 0; i < pairs.length; i++) if (pairs[i][1].toLowerCase() === label || pairs[i][0] === label) return pairs[i][0]; return pairs[0][0]; }

function uniqueName_(ss, wanted, self) {
  var base = String(wanted || 'Продукт').replace(/[\[\]\*\?\/\\:]/g, ' ').trim().slice(0, 90) || 'Продукт';
  var name = base, n = 2;
  while (true) {
    var other = ss.getSheetByName(name);
    if (!other || (self && other.getSheetId() === self.getSheetId())) return name;
    if (name === SUMMARY || name === META) { name = base + ' (' + n++ + ')'; continue; }
    name = base + ' (' + n++ + ')';
  }
}

function writeInputs_(sh, L, p) {
  var inp = p.inputs, ov = p.overrides || {}, i;
  sh.getRange('A1').setValue(clean_(inp.name || 'Продукт'));
  sh.getRange('B2').setValue(clean_(p.brand || ''));
  sh.getRange('A2').setValue('Бренд');
  sh.getRange('A4').setValue(p.note || '').setFontColor('#B4582A');
  sh.getRange('B6:C6').setValues([[inp.unitVolume, inp.unit === 'г' ? 'г' : 'мл']]);
  sh.getRange('B7').setValue(inp.density);
  var r = inp.rates || {};
  sh.getRange('B9:B14').setValues([[inp.batch], [inp.defectPct], [r.USD], [r.EUR], [r.CNY], [!!inp.includeOneOff]]);
  var block = function (row0, cap, rows, width, map) {
    var vals = [];
    for (i = 0; i < cap; i++) { var x = rows[i]; vals.push(x ? map(x) : new Array(width).fill('')); }
    sh.getRange(row0, 1, cap, width).setValues(vals);
  };
  block(L.ing0, L.capIng, inp.ingredients || [], 5, function (x) { return [clean_(x.name), x.pct, x.price, x.cur || 'RUB', x.loss]; });
  block(L.pack0, L.capPack, inp.packaging || [], 4, function (x) { return [clean_(x.name), x.price, x.cur || 'RUB', x.scrap]; });
  block(L.once0, L.capOnce, inp.oneOff || [], 4, function (x) { return [clean_(x.name), x.amount, x.batches, label_(STAGES, x.stage)]; });
  var ovCol = function (row0, cap, list) { var v = []; for (i = 0; i < cap; i++) v.push([list && list[i] !== undefined && list[i] !== null ? list[i] : '']); sh.getRange(row0, 8, cap, 1).setValues(v); };
  ovCol(L.ing0, L.capIng, ov.ing); ovCol(L.pack0, L.capPack, ov.pack);
  var pr = inp.production || {}, mk = inp.marking || {}, lg = inp.logistics || {}, tx = inp.tax || {}, mt = inp.marketing || {};
  sh.getRange(L.fill, 2, 3, 1).setValues([[pr.fillPerUnit], [pr.laborPerUnit], [pr.setupPerBatch]]);
  sh.getRange(L.code, 2, 4, 1).setValues([[mk.codePerUnit], [mk.applyPerUnit], [lg.inboundPerBatch], [lg.fulfilPerUnit]]);
  sh.getRange(L.taxMode, 2, 4, 1).setValues([[label_(TAX_MODES, tx.mode)], [tx.rate], [mt.monthlyBudget], [mt.monthlyUnits]]);
  sh.getRange(L.salesMode, 2).setValue(label_(SALES_MODES, inp.salesMode));
  var s = inp.sales || {}, mp = inp.mp || {};
  var grid = CH_FIELDS.map(function (f) {
    return CHANNELS.map(function (ch) {
      if (ch.id === 'simple') {
        if (SIMPLE_MAP[f[0]]) { var v = s[SIMPLE_MAP[f[0]]]; return f[0] === 'enabled' ? !!v : v; }
        return SIMPLE_FIXED[f[0]];
      }
      var o = mp[ch.id] || {};
      return f[0] === 'enabled' ? !!o.enabled : o[f[0]];
    });
  });
  sh.getRange(L.in0, 2, CH_FIELDS.length, 4).setValues(grid);
}

function push_(ss, p) {
  var meta = readMeta_(ss), byId = {}, i;
  meta.forEach(function (m) { byId[m.id] = m; });
  (p.remove || []).forEach(function (id) {
    var m = byId[id];
    if (m) { var sh = sheetByGid_(ss, m.gid); if (sh) ss.deleteSheet(sh); delete byId[id]; }
  });
  (p.products || []).forEach(function (prod) {
    var m = byId[prod.id], sh = m ? sheetByGid_(ss, m.gid) : null, caps = capsFor_(prod.inputs);
    var rebuild = !sh || !m || m.version !== LAYOUT_VERSION || caps.ing > m.capIng || caps.pack > m.capPack || caps.once > m.capOnce;
    if (!sh) sh = ss.insertSheet(uniqueName_(ss, prod.inputs.name), ss.getSheets().length);
    else { var nm = uniqueName_(ss, prod.inputs.name, sh); if (sh.getName() !== nm) sh.setName(nm); }
    if (!rebuild) caps = { ing: m.capIng, pack: m.capPack, once: m.capOnce };
    var L = layout_(caps.ing, caps.pack, caps.once);
    if (rebuild) build_(sh, L);
    writeInputs_(sh, L, prod);
    byId[prod.id] = { id: prod.id, gid: sh.getSheetId(), name: sh.getName(), capIng: caps.ing, capPack: caps.pack, capOnce: caps.once, version: LAYOUT_VERSION };
  });
  // keep the site order
  var order = p.order || [], list = [];
  order.forEach(function (id) { if (byId[id]) { list.push(byId[id]); delete byId[id]; } });
  Object.keys(byId).forEach(function (id) { list.push(byId[id]); });
  list = list.filter(function (m) { return sheetByGid_(ss, m.gid); });
  list.forEach(function (m) { m.name = sheetByGid_(ss, m.gid).getName(); });
  writeMeta_(ss, list);
  summary_(ss, list);
  bump_(ss);
  SpreadsheetApp.flush();
  return { ok: true, stamp: stamp_(ss), url: ss.getUrl() };
}

// ---------- pull: sheet → site ----------
function pull_(ss, since) {
  var st = stamp_(ss);
  if (since && String(since) === st) return { ok: true, stamp: st, changed: false, url: ss.getUrl() };
  var products = readMeta_(ss).map(function (m) {
    var sh = sheetByGid_(ss, m.gid);
    if (!sh || m.version !== LAYOUT_VERSION) return null;
    var L = layout_(m.capIng, m.capPack, m.capOnce);
    var g = sh.getRange(1, 1, L.end, 8).getValues();
    var v = function (r, c) { return g[r - 1][c - 1]; };
    var rows = function (r0, cap, map) { var out = []; for (var i = 0; i < cap; i++) { var r = r0 + i; if (String(v(r, 1)).trim() !== '') out.push(map(r)); } return out; };
    var ch = {};
    CHANNELS.forEach(function (c, k) {
      var o = {};
      CH_FIELDS.forEach(function (f, j) { o[f[0]] = v(L.in0 + j, k + 2); });
      ch[c.id] = o;
    });
    var simple = {};
    Object.keys(SIMPLE_MAP).forEach(function (f) { simple[SIMPLE_MAP[f]] = ch.simple[f]; });
    return {
      id: m.id,
      inputs: {
        name: v(1, 1), unitVolume: v(6, 2), unit: v(6, 3), density: v(7, 2), batch: v(9, 2), defectPct: v(10, 2),
        rates: { USD: v(11, 2), EUR: v(12, 2), CNY: v(13, 2) }, includeOneOff: v(14, 2) === true,
        ingredients: rows(L.ing0, L.capIng, function (r) { return { name: v(r, 1), pct: v(r, 2), price: v(r, 3), cur: v(r, 4), loss: v(r, 5) }; }),
        packaging: rows(L.pack0, L.capPack, function (r) { return { name: v(r, 1), price: v(r, 2), cur: v(r, 3), scrap: v(r, 4) }; }),
        production: { fillPerUnit: v(L.fill, 2), laborPerUnit: v(L.labor, 2), setupPerBatch: v(L.setup, 2) },
        oneOff: rows(L.once0, L.capOnce, function (r) { return { name: v(r, 1), amount: v(r, 2), batches: v(r, 3), stage: idOf_(STAGES, v(r, 4)) }; }),
        marking: { codePerUnit: v(L.code, 2), applyPerUnit: v(L.apply, 2) },
        logistics: { inboundPerBatch: v(L.inbound, 2), fulfilPerUnit: v(L.fulfil, 2) },
        tax: { mode: idOf_(TAX_MODES, v(L.taxMode, 2)), rate: v(L.taxRate, 2) },
        marketing: { monthlyBudget: v(L.mkBudget, 2), monthlyUnits: v(L.mkUnits, 2) },
        salesMode: idOf_(SALES_MODES, v(L.salesMode, 2)),
        sales: simple,
        mp: { wb: ch.wb, ozon: ch.ozon, ym: ch.ym }
      }
    };
  }).filter(function (x) { return x; });
  return { ok: true, stamp: st, changed: true, products: products, url: ss.getUrl() };
}

// ---------- summary tab ----------
function summary_(ss, list) {
  var sh = ss.getSheetByName(SUMMARY) || ss.insertSheet(SUMMARY, 0);
  sh.clear();
  if (!list.length) { sh.getRange('A1').setValue('Нет продуктов'); return; }
  var q = function (name) { return "'" + name.replace(/'/g, "''") + "'"; };
  var rows = [
    ['Бренд', function () { return 'B2'; }, null, null],
    ['Размер партии, шт.', function () { return 'B9'; }, F_INT, 'SUM'],
    ['СЕБЕСТОИМОСТЬ ЕДИНИЦЫ', function (L) { return 'B' + (L.c + 10); }, F_RUB2, 'AVERAGE'],
    ['Себестоимость без разовых затрат', function (L) { return 'B' + (L.c + 6); }, F_RUB2, 'AVERAGE'],
    ['Себестоимость с разовыми затратами', function (L) { return 'B' + (L.c + 9); }, F_RUB2, 'AVERAGE'],
    ['Сырьё на единицу', function (L) { return 'B' + L.c; }, F_RUB2, 'AVERAGE'],
    ['Упаковка на единицу', function (L) { return 'B' + (L.c + 1); }, F_RUB2, 'AVERAGE'],
    ['Производство на единицу', function (L) { return 'B' + (L.c + 2); }, F_RUB2, 'AVERAGE'],
    ['Разовые затраты на запуск', function (L) { return 'B' + (L.c + 11); }, F_RUB0, 'SUM'],
    ['Деньги на первую партию', function (L) { return 'B' + (L.c + 13); }, F_RUB0, 'SUM'],
    ['Деньги на следующую партию', function (L) { return 'B' + (L.c + 12); }, F_RUB0, 'SUM']
  ];
  CHANNELS.forEach(function (ch, k) {
    var X = COLS[k + 1];
    rows.push(['Цена · ' + ch.label, function (L) { return X + (L.in0 + 1); }, F_RUB0, 'AVERAGE']);
    rows.push(['Прибыль с продажи · ' + ch.label, function (L) { return X + (L.res0 + 10); }, F_RUB2, 'AVERAGE']);
    rows.push(['Маржа · ' + ch.label, function (L) { return X + (L.res0 + 11); }, F_PCT, 'AVERAGE']);
    rows.push(['Прибыль с партии · ' + ch.label, function (L) { return X + (L.res0 + 12); }, F_RUB0, 'SUM']);
  });
  var n = list.length, multi = n > 1;
  var head = ['Показатель'].concat(list.map(function (m) { return m.name; })).concat(multi ? ['Итого / среднее'] : []);
  var data = rows.map(function (row, ri) {
    var line = [row[0]].concat(list.map(function (m) {
      var L = layout_(m.capIng, m.capPack, m.capOnce);
      return '=' + q(m.name) + '!' + row[1](L);
    }));
    if (multi) {
      var rr = ri + 3, from = 'B' + rr, to = COLS[n] ? COLS[n] + rr : columnLetter_(n + 1) + rr;
      line.push(row[3] ? '=IFERROR(' + row[3] + '(' + from + ':' + to + '),"")' : '');
    }
    return line;
  });
  sh.getRange(1, 1).setValue('Сводка по продуктам — обновляется автоматически').setFontWeight('bold').setFontSize(14);
  sh.getRange(2, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground(C_HEAD).setWrap(true);
  sh.getRange(3, 1, data.length, head.length).setValues(data);
  rows.forEach(function (row, ri) { if (row[2]) sh.getRange(ri + 3, 2, 1, head.length - 1).setNumberFormat(row[2]); });
  sh.getRange(5, 1, 1, head.length).setFontWeight('bold').setBackground(C_KEY);
  if (multi) sh.getRange(3, head.length, data.length, 1).setBackground(C_SUB).setFontWeight('bold');
  sh.setColumnWidth(1, 300);
  for (var c = 2; c <= head.length; c++) sh.setColumnWidth(c, 150);
  sh.setFrozenRows(2); sh.setFrozenColumns(1);
}
function columnLetter_(n) { var s = ''; while (n > 0) { var m = (n - 1) % 26; s = String.fromCharCode(65 + m) + s; n = Math.floor((n - 1) / 26); } return s; }
